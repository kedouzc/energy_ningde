// 1. 添加老虎机跳动效果
const [animatedBounty, setAnimatedBounty] = useState(previewBounty);

useEffect(() => {
  // 数字滚动动画
  const duration = 500;
  const startValue = animatedBounty;
  const endValue = previewBounty;
  const startTime = performance.now();
  
  const animate = (currentTime: number) => {
    const elapsed = currentTime - startTime;
    const progress = Math.min(elapsed / duration, 1);
    const easeOutQuart = 1 - Math.pow(1 - progress, 4);
    
    setAnimatedBounty(startValue + (endValue - startValue) * easeOutQuart);
    
    if (progress < 1) requestAnimationFrame(animate);
  };
  
  requestAnimationFrame(animate);
}, [previewBounty]);

const handleMixChange = (style: keyof typeof mix, value: number) => {
  const newHistory = [style, ...sliderHistory.filter(s => s !== style)];
  setSliderHistory(newHistory);
  
  const first = newHistory[0] as keyof typeof mix;  // 当前滑动的
  const second = newHistory[1] as keyof typeof mix; // 之前滑过，已锁定
  const third = newHistory[2] as keyof typeof mix;  // 自动联动的
  
  let newMix = { ...mix, [first]: value };
  
  // 第二个滑块保持不动
  newMix[second] = mix[second];
  
  // 第三个滑块自动调整，保持总和 100%
  newMix[third] = 100 - value - newMix[second];
  
  // 边界检查
  if (newMix[third] < 0) {
    newMix[third] = 0;
    newMix[second] = 100 - value;
